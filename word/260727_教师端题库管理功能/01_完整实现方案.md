## 1. 需求概述

### 1.1 核心需求

重新开发教师端题库管理功能，创建一个全新的一体化页面，参考截图中的布局风格（左侧知识树目录 + 中间题目详情展示 + 右侧操作按钮），支持以下三种添加题目方式：

| 方式 | 说明 | 现有支持 |
|------|------|----------|
| **方式1: 拍照/图片导入** | 保持现有拍照/AI识别功能不变 | ✅ 已有 |
| **方式2: PDF/Word 导入** | 上传PDF/Word文件，后端解析提取题目 | ✅ 已有 |
| **方式3: JSON数据包导入** | 上传zip压缩包（JSON+PNG图片），按预定义格式解析导入 | ❌ 新增 |

### 1.2 页面设计要求

参考截图布局，采用三栏式设计：
- **顶部工具栏**: 题目/知识/年份/标题/题号/标签/知识点/关键词查询/排序/ID/相似题/重置 + 翻页
- **左侧目录树**: 学科选择 + 搜索 + 目录管理（树形结构：年级→学期→章节→知识点）
- **中间内容区**: 题目详细展示（题干、图片、公式、选项、子问题、答案、解析），支持上下翻页
- **右侧操作区**: 随机选题/查询参数/刷新题目/显示答案/分享多题/分享历史/加入篮子

### 4.3 新增 API 封装

#### `api/questions.ts` — 新增接口

```typescript
// 新增: JSON数据包导入
export function importJsonPackage(formData: FormData) {
  return new Promise<any>((resolve, reject) => {
    const token = uni.getStorageSync('accessToken')
    // #ifdef H5
    fetch(`${UPLOAD_BASE}/questions/import-json-package`, {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${token}` },
      body: formData,
    })
      .then(res => res.json())
      .then(data => data.code === 0 ? resolve(data) : reject(new Error(data.message)))
      .catch(reject)
    // #endif
    // #ifndef H5
    uni.uploadFile({
      url: `${UPLOAD_BASE}/questions/import-json-package`,
      filePath: (formData as any).__file_path__,
      name: 'file',
      header: { 'Authorization': `Bearer ${token}` },
      success: (res) => {
        const data = JSON.parse(res.data)
        data.code === 0 ? resolve(data) : reject(new Error(data.message))
      },
      fail: reject,
    })
    // #endif
  })
}

// 新增: 获取导入任务进度
export function getImportTaskStatus(taskId: string) {
  return get<any>(`/questions/import-json-task/${taskId}/status/`)
}

// 新增: 获取导入历史
export function getJsonImportHistory() {
  return get<any>('/questions/json-import-history/')
}

// 新增: 批量操作接口
export function batchUpdateQuestions(data: {
  ids: string[]
  action: string  // 'confirm' | 'reject' | 'delete' | 'add_to_favorites' | 'set_difficulty' | 'set_kp'
  value?: any
}) {
  return post<any>('/questions/batch-update/', data)
}

// 新增: 题目篮子（购物车）
export function getBasket() {
  return get<any>('/questions/basket/')
}
export function addToBasket(questionId: string) {
  return post<any>('/questions/basket/add/', { question_id: questionId })
}
export function removeFromBasket(questionId: string) {
  return del(`/questions/basket/${questionId}/`)
}
export function clearBasket() {
  return del('/questions/basket/clear/')
}
```

```python
"""JSON数据包导入视图：上传ZIP → 解析 → 导入题目和图片"""
import os
import json
import uuid
import zipfile
import shutil
import logging
from django.conf import settings
from django.db import transaction
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response

from apps.parser.models import ExamQuestion, QuestionOption, QuestionImage
from apps.papers.models import ExamPaper, ParseTask
from apps.common.codegen import generate_question_system_id

logger = logging.getLogger(__name__)

QUESTION_TYPE_MAP = {
    'single_choice': 'single_choice',
    'multiple_choice': 'multiple_choice',
    'fill_blank': 'fill_blank',
    'short_answer': 'short_answer',
    'essay': 'essay',
    'true_false': 'true_false',
    'computation': 'computation',
    'proof': 'proof',
    'solution': 'short_answer',  # 解答题映射为简答题
    # ✅ 真实数据包中出现的题型
    'calculation': 'computation',           # 计算题→计算题
    'experiment': 'short_answer',           # 实验探究题→简答题
    'reading_comprehension': 'short_answer', # 阅读理解题→简答题
    'unknown': 'unknown',
}

# 学科映射：中文/英文 → 字母代码（用于 generate_question_system_id）
SUBJECT_MAP = {
    '数学': 'M', 'math': 'M',
    '物理': 'P', 'physics': 'P',
    '化学': 'C', 'chemistry': 'C',
    '语文': 'Z', 'chinese': 'Z',
    '英语': 'E', 'english': 'E',
}


@api_view(['POST'])
@permission_classes([IsAuthenticated])
def import_json_package(request):
    """上传ZIP压缩包，解析JSON和图片，批量导入题目。"""
    uploaded_file = request.FILES.get('file')
    if not uploaded_file:
        return Response({'code': 400, 'message': '请上传ZIP文件', 'data': None}, status=400)

    if not uploaded_file.name.lower().endswith('.zip'):
        return Response({'code': 400, 'message': '仅支持 .zip 格式文件', 'data': None}, status=400)

    # 保存到临时目录
    temp_dir = settings.MEDIA_ROOT / 'temp_imports' / str(uuid.uuid4())
    temp_dir.mkdir(parents=True, exist_ok=True)
    zip_path = temp_dir / uploaded_file.name

    with open(zip_path, 'wb') as f:
        for chunk in uploaded_file.chunks():
            f.write(chunk)

    # 异步解析导入
    task = ParseTask.objects.create(
        paper=None,  # JSON导入无对应ExamPaper，后续创建
        task_type='json_import',
        status='running',
        progress=0,
        current_step='正在解压文件',
    )

    try:
        result = _process_json_import(zip_path, temp_dir, task, request.user)
        task.status = 'success'
        task.progress = 100
        task.current_step = '导入完成'
        task.save()
        return Response({
            'code': 0, 'message': '导入成功',
            'data': result,
        })
    except Exception as e:
        logger.exception(f'JSON import failed: {e}')
        task.status = 'failed'
        task.error_message = str(e)
        task.save()
        return Response({'code': 500, 'message': f'导入失败: {str(e)}', 'data': None}, status=500)
    finally:
        # ✅ 修复: 清理临时文件，防止磁盘空间累积
        shutil.rmtree(temp_dir, ignore_errors=True)


def _process_json_import(zip_path, temp_dir, task, user):
    """处理JSON导入的核心逻辑。"""
    # 1. 解压ZIP
    with zipfile.ZipFile(zip_path, 'r') as zf:
        zf.extractall(temp_dir)

    # 2. 查找 all_questions.json 或 manifest.json
    all_questions_path = temp_dir / 'all_questions.json'
    manifest_path = temp_dir / 'manifest.json'

    questions_data = []
    paper_info = None

    if all_questions_path.exists():
        with open(all_questions_path, 'r', encoding='utf-8') as f:
            package = json.load(f)
        paper_info = package.get('paper', {})
        questions_data = package.get('questions', [])
    elif manifest_path.exists():
        with open(manifest_path, 'r', encoding='utf-8') as f:
            manifest = json.load(f)

        # 优先使用 combined_json
        combined = manifest.get('combined_json')
        if combined:
            combined_path = temp_dir / combined
            if combined_path.exists():
                with open(combined_path, 'r', encoding='utf-8') as f:
                    package = json.load(f)
                paper_info = package.get('paper', manifest)
                questions_data = package.get('questions', [])

        # 如果没有 combined_json，使用 files 列表逐个读取
        if not questions_data:
            for qfile in manifest.get('files', []):
                qpath = temp_dir / qfile
                if qpath.exists():
                    with open(qpath, 'r', encoding='utf-8') as f:
                        questions_data.append(json.load(f))

    if not questions_data:
        raise ValueError('未找到有效的题目数据')

    # 3. 创建 ExamPaper 记录
    # 3. 创建 ExamPaper
    # ✅ 修复: 存中文名称（如'物理'）而非字母代码，因为 pre_save signal 的
    # generate_paper_code 使用 SUBJECT_FIRST_LETTER 字典（key是中文名称）
    subject_name = paper_info.get('subject', '物理')

    paper = ExamPaper.objects.create(
        title=paper_info.get('title', 'JSON导入试卷'),
        subject=subject_name,  # '物理'，由signal自动生成paper_code
        grade=paper_info.get('grade', ''),
        stage='junior' if '年级' in paper_info.get('grade', '') else 'senior',
        paper_type='json_import',
        source_file_path=str(zip_path.relative_to(settings.MEDIA_ROOT)),
        status='reviewing',
        uploaded_by=user,
    )

    # 4. 逐题导入
    task.progress = 10
    task.save()

    imported_count = 0
    error_count = 0
    errors = []

    total = len(questions_data)
    assets_dir = temp_dir / 'assets'

    for i, qdata in enumerate(questions_data):
        try:
            _import_single_question(qdata, paper, assets_dir, temp_dir)
            imported_count += 1
        except Exception as e:
            error_count += 1
            errors.append({
                'question_no': qdata.get('question_no', '?'),
                'error': str(e),
            })
            logger.warning(f'Failed to import question {qdata.get("question_no")}: {e}')

        # 更新进度
        task.progress = 10 + int(85 * (i + 1) / total)
        task.current_step = f'正在导入第 {i + 1}/{total} 题'
        task.save()

    # 5. 更新试卷统计
    paper.total_questions = imported_count
    paper.save(update_fields=['total_questions'])

    return {
        'paper_id': str(paper.id),
        'imported': imported_count,
        'errors': error_count,
        'error_details': errors[:10],  # 最多返回10条错误
    }


def _import_single_question(qdata, paper, assets_dir, base_dir):
    """导入单道题目及其关联图片。"""
    qtype = QUESTION_TYPE_MAP.get(qdata.get('question_type', 'unknown'), 'unknown')
    system_id = generate_question_system_id(SUBJECT_MAP.get(paper.subject, 'P'))

    # 创建题目
    question = ExamQuestion.objects.create(
        paper=paper,
        system_id=system_id,
        question_no=str(qdata.get('question_no', '')),
        paper_question_no=f"{paper.paper_code or 'JSON'}-{qdata.get('section', '')}-{qdata.get('question_no', '')}",
        question_type=qtype,
        section_title=qdata.get('section', ''),
        stem=qdata.get('stem', ''),
        answer=qdata.get('answer', {}).get('raw', '') if isinstance(qdata.get('answer'), dict) else str(qdata.get('answer', '')),
        analysis=qdata.get('analysis', ''),
        solution='',
        knowledge_points=_extract_knowledge_points(qdata),
        difficulty=_extract_difficulty(qdata),
        page_start=qdata.get('source', {}).get('page_start', 0),
        page_end=qdata.get('source', {}).get('page_end', 0),
        confidence=1.0,
        need_review=True,
        review_status='need_review',
        parse_status='json_imported',
    )

    # 创建选项
    for opt in qdata.get('options', []):
        QuestionOption.objects.create(
            question=question,
            option_label=opt.get('label', 'A'),
            content=opt.get('content', ''),
            sort_order=ord(opt.get('label', 'A')) - 65,
        )

    # 导入插图
    for ill in qdata.get('illustrations', []):
        _import_image(ill, question, paper, assets_dir, image_type='diagram')

    # 导入公式图片
    for fa in qdata.get('formula_assets', []):
        _import_image(fa, question, paper, assets_dir, image_type='formula')


def _import_image(img_data, question, paper, assets_dir, image_type='other'):
    """导入单张图片，复制到 media 目录。"""
    file_rel = img_data.get('file', '')
    if not file_rel:
        return

    # 解析图片路径（相对路径）
    src_path = assets_dir / file_rel.replace('../assets/', '')
    if not src_path.exists():
        # 尝试直接拼接
        src_path = assets_dir / os.path.basename(file_rel)

    if not src_path.exists():
        logger.warning(f'Image not found: {file_rel}')
        return

    # 复制到 media 目录
    dest_dir = settings.MEDIA_ROOT / 'exams' / str(paper.id) / 'json_images'
    dest_dir.mkdir(parents=True, exist_ok=True)
    dest_name = f'{question.id}_{os.path.basename(src_path)}'
    dest_path = dest_dir / dest_name

    import shutil
    shutil.copy2(str(src_path), str(dest_path))

    rel_media = f'exams/{paper.id}/json_images/{dest_name}'

    QuestionImage.objects.create(
        paper=paper,
        question=question,
        image_type=image_type,
        file_path=rel_media,
        description=img_data.get('alt_text', img_data.get('recognized_text', '')),
        sort_order=0,
    )


def _extract_knowledge_points(qdata):
    """从JSON数据提取知识点。"""
    # JSON格式中没有直接的知识点，返回空列表，后续由AI或教师手动标注
    return []


def _extract_difficulty(qdata):
    """从JSON数据提取难度。"""
    # JSON格式中没有直接的难度字段，默认3
    return 3.0
```
